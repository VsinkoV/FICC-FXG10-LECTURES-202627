  \begin{outlook}{1 to 3 month outlook}{Bullish USD: markets underestimate how far the Fed will tighten.}
    \point{Priced}{72\% odds of a 25bp hike on 28 Oct ($\approx$+18bp; CME FedWatch, 28 Sep), yet only [about 30\%] for a second by December. The Fed's median implies one more hike, then a pause at 4.1\%.}
    \point{View}{\textbf{The Fed out-hawks market pricing.}
      \evidence{Fed}{16 of 18 officials expect a further hike; the 2026 median climbed from 3.8\% to 4.1\% (SEP, 16 Sep); Warsh: inflation ``too high for too long''.}
      \evidence{Inflation}{CPI 3.4\%, core +0.3\% m/m (BLS, Aug); Brent near \$107 and consumer expectations of 4.6\% (UMich, Sep) add pressure.}
      \evidence{Jobs}{payrolls +162k vs +53k expected, unemployment 4.1\% (BLS, Aug): the economy can absorb tighter policy.}}
    \point{Positioning}{DXY net longs +10,330 (CFTC, 22 Sep), z $\approx$ +1: modestly long, far from crowded. Yet euro shorts nearly doubled to $-$52,334: our trade is growing popular.}
    \point{Change our mind}{Payrolls below 50k (2 Oct), core CPI $\leq$0.2\% m/m (14 Oct), a hold on 28 Oct, or an Iran ceasefire sending Brent under \$90. Any dovish surprise risks a euro short squeeze.}
  \end{outlook}
  \vspace{3pt}
%% 6 to 12 months = FUNDAMENTALS: the five structural drivers (Lecture 3).
  \begin{outlook}{6 to 12 month outlook}{Energy and El Ni\~no tilt the balance: the dollar stays mildly bid, though rich valuation caps the upside.}
    \point{Priced}{Rates: about 4.3\% by December, hikes into 2027. Oil: Brent falling from \$107 to the \$70s by 2027 (EIA).}
    \point{View}{\textbf{Mildly bullish USD:} the supports are immediate, the drags slow.
      \evidence{Real rates (+)}{real policy rate +0.5\%, a 1.54 pt premium over the DXY basket.}
      \evidence{Terms of trade (+)}{the US exports energy, Europe imports it: EU gas storage is a record-low 71\% (27 Sep), diesel stocks the thinnest since 2022. El Ni\~no ($>$90\%, NOAA) also hits JPY and AUD.}
      \evidence{Balance of payments ($-$)}{deficit widened to 3.0\% of GDP (BEA, Q2) as foreign inflows slowed to \$83.7bn in July from \$133.5bn (TIC).}
      \evidence{Valuation ($-$)}{REER 114, near its record; about 17\% overvalued against EUR on PPP.}}
    \point{Where we disagree}{With Europe's gas and diesel stocks near record lows, energy stays tighter for longer than futures imply, weighing on the euro more than the dollar.}
    \point{Change our mind}{A mild winter or Iran ceasefire collapsing energy prices; or growth below 1\% forcing Fed cuts.}
  \end{outlook}
